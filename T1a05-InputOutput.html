function mySquareFunction() {
    let myIn01 = parseFloat(
        document.getElementById('myNum01').value
    );

    if (document.getElementById('myNum01').value.trim() === ''
        || !Number.isFinite(myIn01)) {
        document.getElementById('myDiv01').innerHTML =
            'Please enter a valid number.';
        return;
    }

    let myAnswer = myIn01 * myIn01;

    document.getElementById('myDiv01').innerHTML =
        myIn01 + ' squared = ' + myAnswer;
}
